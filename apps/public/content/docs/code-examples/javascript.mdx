## JavaScript / Node.js

JavaScript is one of the most popular languages for interacting with REST APIs. The JKT48Connect API is easy to access from both the client-side (browser) and the server-side (Node.js).

### Using the native `fetch` API

This is the simplest, most standardized way, requiring no external libraries:

```javascript
const API_KEY = 'YOUR_API_KEY';
const BASE_URL = 'https://v2.jkt48connect.com';

async function getJKT48Data(endpoint) {
  try {
    const response = await fetch(`${BASE_URL}/api/jkt48/${endpoint}?apikey=${API_KEY}`);
    
    if (!response.ok) {
      throw new Error(`Error: ${response.status} - ${response.statusText}`);
    }
    
    return await response.json();
  } catch (error) {
    console.error('An error occurred:', error.message);
    return null;
  }
}

// Executing the function
(async () => {
  // Fetch member data
  const members = await getJKT48Data('members');
  console.log(`Found ${members?.length || 0} members.`);
})();
```

### Using `axios` (Node.js / React)

If you use the `axios` library, the process is very straightforward:

```javascript
// npm install axios
const axios = require('axios');

const API_KEY = 'YOUR_API_KEY';

axios.get('https://v2.jkt48connect.com/api/jkt48/live', {
  params: {
    apikey: API_KEY
  }
})
.then(response => {
  const liveMembers = response.data;
  console.log('Members currently live:', liveMembers.map(m => m.name));
})
.catch(error => {
  console.error('API Error:', error.response?.data?.message || error.message);
});
```

### Example in React.js

Here is a sample React component that displays the theater schedule data:

```jsx
import React, { useState, useEffect } from 'react';

const API_KEY = process.env.REACT_APP_JKT48_API_KEY;

export default function TheaterSchedule() {
  const [schedule, setSchedule] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch(`https://v2.jkt48connect.com/api/jkt48/theater?apikey=${API_KEY}`)
      .then(res => res.json())
      .then(data => {
        // Filter upcoming shows only
        setSchedule(data.filter(s => s.is_upcoming));
        setLoading(false);
      })
      .catch(err => {
        console.error(err);
        setLoading(false);
      });
  }, []);

  if (loading) return <p>Loading theater schedule...</p>;

  return (
    <div className="theater-schedule">
      <h2>🎭 JKT48 Theater Schedule</h2>
      {schedule.length === 0 ? (
        <p>No schedules announced yet.</p>
      ) : (
        <ul>
          {schedule.map((show, idx) => (
            <li key={idx}>
              <strong>{show.title} ({show.setlist})</strong> - {show.date} {show.time}
              <br/>
              Performing: {show.members.join(', ')}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

> **Security Note**: If you call the API directly from a front-end React application, your API Key could be exposed to users. Ensure you understand the risks or use a back-end as a secure intermediary layer.
